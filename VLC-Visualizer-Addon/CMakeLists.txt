cmake_minimum_required(VERSION 3.24)
project(VLCVisualizerAddon VERSION 0.5.0 LANGUAGES C CXX)
option(VIS_BUILD_DAWN_RUNTIME "Build the shared native Dawn runtime beside the original hosts" OFF)
if(UNIX)
  add_subdirectory(native)
  if(VIS_BUILD_DAWN_RUNTIME)
    set(VIS_BUILD_NATIVE_WINDOW ON CACHE BOOL "" FORCE)
    add_subdirectory(dawn)
  endif()
  return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build this project for Windows AMD64 with MSVC")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX)
include_directories(include vendor ports)
add_executable(Visualizers WIN32 src/host.cpp)
target_link_libraries(Visualizers PRIVATE comctl32 shell32 shlwapi winmm d3d11 dxgi d3dcompiler psapi)
add_executable(VlcVisualizerLauncher WIN32 src/launcher.cpp)
target_link_libraries(VlcVisualizerLauncher PRIVATE shell32 version)
add_executable(VisualizersCtl src/addonctl.cpp)
add_executable(package_install_tests tests/package_install_tests.cpp)
target_link_libraries(package_install_tests PRIVATE bcrypt)
add_library(qt_bootstrap MODULE src/qt_bootstrap.cpp)
set_target_properties(qt_bootstrap PROPERTIES PREFIX "lib" OUTPUT_NAME "qt_plugin")
target_link_libraries(VisualizersCtl PRIVATE bcrypt version shell32 user32)
add_library(vlc_visualizers MODULE src/vlc_audio.cpp src/vlc_menu.cpp)
target_include_directories(vlc_visualizers PRIVATE vendor/vlc/include)
target_compile_definitions(vlc_visualizers PRIVATE __PLUGIN__ MODULE_STRING="vlc_visualizers" __VLC__)
# One argument lets each generator escape paths, including spaces, correctly.
target_compile_options(vlc_visualizers PRIVATE
  "$<$<COMPILE_LANGUAGE:CXX>:/FI${CMAKE_CURRENT_SOURCE_DIR}/ports/vlc_msvc.h>")
target_link_libraries(vlc_visualizers PRIVATE comctl32 shell32)
set_target_properties(vlc_visualizers PROPERTIES PREFIX "lib" OUTPUT_NAME "vlc_visualizers_plugin")
add_executable(core_tests tests/core_tests.cpp)
add_executable(display_probe tests/display_probe.cpp)
target_link_libraries(display_probe PRIVATE dxgi user32)
enable_testing()
add_test(NAME audio_core COMMAND core_tests)
if(VIS_BUILD_DAWN_RUNTIME)
  set(VIS_BUILD_NATIVE_WINDOW ON CACHE BOOL "" FORCE)
  add_subdirectory(dawn)
endif()
add_executable(renderer_probe tests/renderer_probe.cpp)
target_link_libraries(renderer_probe PRIVATE psapi)
option(ENABLE_ASAN_TESTS "Instrument the renderer probe with AddressSanitizer" OFF)
if(ENABLE_ASAN_TESTS)
  set_property(TARGET renderer_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  target_compile_options(renderer_probe PRIVATE /fsanitize=address /Zi)
  target_link_options(renderer_probe PRIVATE /INCREMENTAL:NO /DEBUG)
  add_library(asan_sanity_dll MODULE tests/asan_dll_sanity.cpp)
  add_executable(asan_sanity tests/asan_dll_sanity.cpp)
  target_compile_definitions(asan_sanity_dll PRIVATE SANITY_DLL)
  foreach(target asan_sanity asan_sanity_dll)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
    target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO /DEBUG)
  endforeach()
endif()
