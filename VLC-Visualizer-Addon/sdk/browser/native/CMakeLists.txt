cmake_minimum_required(VERSION 3.24)
project(VisualizerAudioBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
add_library(visualizer_audio MODULE bridge.cpp)
set_target_properties(visualizer_audio PROPERTIES PREFIX "" SUFFIX ".node")
set(AUDIO_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../include" CACHE PATH "Shared audio SDK headers")
target_include_directories(visualizer_audio PRIVATE vendor "${AUDIO_INCLUDE_DIR}")
target_compile_definitions(visualizer_audio PRIVATE NAPI_VERSION=8 NOMINMAX _CRT_SECURE_NO_WARNINGS)
if(WIN32)
  target_sources(visualizer_audio PRIVATE delay_hook.cpp)
  set(NODE_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/browser-runtime/node.lib" CACHE FILEPATH "Electron Node import library")
  target_link_libraries(visualizer_audio PRIVATE "${NODE_LIBRARY}" delayimp)
  target_link_options(visualizer_audio PRIVATE /DELAYLOAD:node.exe)
elseif(APPLE)
  target_link_options(visualizer_audio PRIVATE "-undefined" "dynamic_lookup")
else()
  target_link_libraries(visualizer_audio PRIVATE pthread)
endif()
