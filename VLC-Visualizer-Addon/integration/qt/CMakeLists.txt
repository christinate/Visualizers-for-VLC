cmake_minimum_required(VERSION 3.24)
project(VlcInlineVisualizers LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC OFF)
set(CMAKE_AUTOUIC OFF)
set(CMAKE_AUTORCC ON)
set(VLC_QT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/vlc/qt" CACHE PATH "VLC 3 Qt source directory")
set(VLC_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../vendor/vlc/include" CACHE PATH "VLC plugin headers")
set(VLC_CORE_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/vlc-imports/libvlccore.lib" CACHE FILEPATH "Matching VLC core import/shared library")
set(VLC_PUBLIC_LIBRARY "${CMAKE_CURRENT_SOURCE_DIR}/../../build/vlc-imports/libvlc.lib" CACHE FILEPATH "Matching LibVLC import/shared library")
if(WIN32)
  if(MSVC)
    message(FATAL_ERROR "Build VLC's Qt interface with the matching MinGW/Qt toolchain. The host/renderers still use MSVC; the GUI must match VLC's C runtime.")
  endif()
  find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Svg Network)
  set(QT Qt6)
else()
  find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets Svg Network)
  set(QT Qt5)
endif()
include(sources.cmake)
file(GLOB UI_FORMS "${VLC_QT_SOURCE}/ui/*.ui")
set(UI_HEADERS)
foreach(form IN LISTS UI_FORMS)
  get_filename_component(name "${form}" NAME_WE)
  set(output "${CMAKE_CURRENT_BINARY_DIR}/generated/ui/${name}.h")
  add_custom_command(OUTPUT "${output}" COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_CURRENT_BINARY_DIR}/generated/ui" COMMAND $<TARGET_FILE:${QT}::uic> "${form}" -o "${output}" DEPENDS "${form}" VERBATIM)
  list(APPEND UI_HEADERS "${output}")
endforeach()
add_library(vlc_qt_inline MODULE ${VLC_QT_FILES} ${UI_HEADERS} "${VLC_QT_SOURCE}/vlc.qrc")
target_sources(vlc_qt_inline PRIVATE "${VLC_QT_SOURCE}/dialogs/vlm.cpp")
set_target_properties(vlc_qt_inline PROPERTIES PREFIX "lib" OUTPUT_NAME "qt_plugin" AUTOUIC_SEARCH_PATHS "${VLC_QT_SOURCE}/ui")
target_include_directories(vlc_qt_inline PRIVATE "${VLC_INCLUDE_DIR}" "${VLC_QT_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_compile_definitions(vlc_qt_inline PRIVATE __PLUGIN__ MODULE_STRING="qt" __VLC__ QT_USE_QSTRINGBUILDER QT_NO_CAST_FROM_BYTEARRAY HAVE_CONFIG_H)
target_link_libraries(vlc_qt_inline PRIVATE ${QT}::Core ${QT}::Gui ${QT}::Widgets ${QT}::Svg ${QT}::Network "${VLC_CORE_LIBRARY}" "${VLC_PUBLIC_LIBRARY}")
file(GLOB_RECURSE QT_HEADERS "${VLC_QT_SOURCE}/*.hpp")
set(MOC_HEADERS)
foreach(header IN LISTS QT_HEADERS)
  file(READ "${header}" content)
  if(content MATCHES "Q_OBJECT")
    list(APPEND MOC_HEADERS "${header}")
  endif()
endforeach()
if(WIN32)
  qt6_wrap_cpp(MOC_SOURCES ${MOC_HEADERS} TARGET vlc_qt_inline)
else()
  qt5_wrap_cpp(MOC_SOURCES ${MOC_HEADERS} TARGET vlc_qt_inline)
endif()
target_sources(vlc_qt_inline PRIVATE ${MOC_SOURCES})
if(WIN32)
  target_sources(vlc_qt_inline PRIVATE "${VLC_QT_SOURCE}/main_interface_win32.cpp" "${VLC_QT_SOURCE}/main_interface_win32.hpp")
  target_link_libraries(vlc_qt_inline PRIVATE Qt6::GuiPrivate comctl32 ole32 uuid shell32 shlwapi winmm ws2_32)
  target_compile_definitions(vlc_qt_inline PRIVATE NOMINMAX _WIN32_WINNT=0x0601)
endif()
