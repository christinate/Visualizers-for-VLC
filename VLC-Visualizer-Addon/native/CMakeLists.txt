cmake_minimum_required(VERSION 3.24)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(../sdk/native native-runtime)
set(VLC_PLUGIN_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../vendor/vlc/include" CACHE PATH "Matching VLC plugin headers")
if(APPLE)
  set(VLC_CORE_LIBRARY "/Applications/VLC.app/Contents/MacOS/lib/libvlccore.dylib" CACHE FILEPATH "VLC core")
  enable_language(OBJCXX)
  add_library(vlc_visualizers MODULE vlc_bridge.cpp menu_mac.mm)
  option(VIS_NATIVE_MENU_TEST_CONTROL "Compile owner-local AppKit menu integration fixtures; never enable for release packages" OFF)
  if(VIS_NATIVE_MENU_TEST_CONTROL)
    target_compile_definitions(vlc_visualizers PRIVATE VIS_NATIVE_MENU_TEST_CONTROL)
  endif()
  target_link_libraries(vlc_visualizers PRIVATE "${VLC_CORE_LIBRARY}" "-framework AppKit")
  set_target_properties(vlc_visualizers PROPERTIES SUFFIX ".dylib")
else()
  set(VLC_QT_MAJOR 5 CACHE STRING "Qt major version used by the installed VLC interface")
  find_package(Qt${VLC_QT_MAJOR} REQUIRED COMPONENTS Core Gui Widgets)
  find_library(VLC_CORE_LIBRARY NAMES vlccore libvlccore.so.9 PATHS /usr/lib/x86_64-linux-gnu REQUIRED)
  add_library(vlc_visualizers MODULE vlc_bridge.cpp menu_qt.cpp)
  target_link_libraries(vlc_visualizers PRIVATE "${VLC_CORE_LIBRARY}" Qt${VLC_QT_MAJOR}::Core Qt${VLC_QT_MAJOR}::Gui Qt${VLC_QT_MAJOR}::Widgets)
endif()
target_include_directories(vlc_visualizers PRIVATE "${VLC_PLUGIN_INCLUDE}" ../sdk/native ../include ../sdk/runtime/vendor ../integration/qt)
target_compile_definitions(vlc_visualizers PRIVATE __PLUGIN__ MODULE_STRING="vlc_visualizers" __VLC__)
set_target_properties(vlc_visualizers PROPERTIES PREFIX "lib" OUTPUT_NAME "vlc_visualizers_plugin")
if(NOT APPLE)
  set_target_properties(vlc_visualizers PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../lib")
endif()
install(TARGETS vlc_visualizers LIBRARY DESTINATION vlc-plugins)
