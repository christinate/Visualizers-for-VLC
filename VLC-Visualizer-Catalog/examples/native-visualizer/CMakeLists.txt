cmake_minimum_required(VERSION 3.24)
project(ExampleNativeVisualizer LANGUAGES CXX)
add_library(vis_example MODULE plugin.cpp)
set_target_properties(vis_example PROPERTIES PREFIX "" CXX_STANDARD 17)
if(APPLE)
  target_link_libraries(vis_example PRIVATE "-framework OpenGL")
  set_target_properties(vis_example PROPERTIES SUFFIX ".dylib")
else()
  find_package(OpenGL REQUIRED)
  target_link_libraries(vis_example PRIVATE OpenGL::GL)
endif()
