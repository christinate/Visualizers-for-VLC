cmake_minimum_required(VERSION 3.24)
project(AudioColorExample VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use a 64-bit target")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
set(ADDON_SOURCE "" CACHE PATH "Matching VLC-Visualizer-Addon source checkout")
if(NOT EXISTS "${ADDON_SOURCE}/sdk/native/visualizer_dawn_api.h")
  message(FATAL_ERROR "Provide ADDON_SOURCE for the API 4 header")
endif()
find_package(Dawn CONFIG REQUIRED)
get_target_property(runtime_type dawn::webgpu_dawn TYPE)
if(NOT runtime_type STREQUAL "SHARED_LIBRARY")
  message(FATAL_ERROR "Link the suite's pinned shared Dawn; do not embed a private implementation")
endif()
file(READ "${ADDON_SOURCE}/dawn/dependencies.lock.json" dependencies)
string(JSON DAWN_REVISION GET "${dependencies}" source_revision)
if(NOT DAWN_REVISION STREQUAL "7daf3b0bee2c73621001ab15b256b5a84fb0b45f")
  message(FATAL_ERROR "Update the example and manifest schema when the native ABI pin changes")
endif()
add_library(audio_color MODULE plugin.cpp)
target_include_directories(audio_color PRIVATE "${ADDON_SOURCE}/sdk/native")
target_link_libraries(audio_color PRIVATE dawn::webgpu_dawn)
set_target_properties(audio_color PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/audio_color"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/package/audio_color")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(audio_color PROPERTIES LIBRARY_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/package/audio_color"
    RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/package/audio_color")
endforeach()
if(WIN32)
  set(EXAMPLE_PLATFORM windows)
  set(EXAMPLE_ARCH x86_64)
elseif(APPLE)
  set(EXAMPLE_PLATFORM macos)
  set(EXAMPLE_ARCH arm64)
else()
  set(EXAMPLE_PLATFORM linux)
  set(EXAMPLE_ARCH x86_64)
  set_target_properties(audio_color PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../../native")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/package/audio_color/assets")
configure_file(manifest.json.in "${CMAKE_BINARY_DIR}/package/audio_color/manifest.json" @ONLY)
configure_file(assets/README.txt "${CMAKE_BINARY_DIR}/package/audio_color/assets/README.txt" COPYONLY)
configure_file(LICENSE.txt "${CMAKE_BINARY_DIR}/package/audio_color/LICENSE.txt" COPYONLY)
