cmake_minimum_required(VERSION 3.24)
project(Geiss64 VERSION 0.5.0 LANGUAGES CXX)
option(VIS_BUILD_DAWN_RUNTIME "Build the native runtime and Prism Flow module from the pinned SDK" OFF)
if(UNIX)
  add_subdirectory(sdk/native)
  add_subdirectory(native)
  if(VIS_BUILD_DAWN_RUNTIME)
    add_subdirectory(sdk/dawn)
    add_subdirectory(native/dawn)
  endif()
  return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Geiss rendering requires Windows AMD64/MSVC. Build Winamp bridges separately.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
option(ENABLE_ASAN "Enable AddressSanitizer for renderer validation" OFF)
add_subdirectory(sdk/runtime)
add_library(vis_geiss64 MODULE source/main.cpp source/helper.cpp ports/geiss_map.cpp)
target_compile_definitions(vis_geiss64 PRIVATE PLUGIN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(vis_geiss64 PRIVATE
  "$<$<COMPILE_LANGUAGE:CXX>:/FI${CMAKE_CURRENT_SOURCE_DIR}/ports/legacy_windows.h>"
  /wd4244 /wd4267)
target_include_directories(vis_geiss64 PRIVATE source)
target_link_libraries(vis_geiss64 PRIVATE comctl32 ddraw dsound dxguid winmm shell32 advapi32)
if(ENABLE_ASAN)
  set_property(TARGET vis_geiss64 PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  target_compile_options(vis_geiss64 PRIVATE /fsanitize=address /Zi)
  target_link_options(vis_geiss64 PRIVATE /INCREMENTAL:NO /DEBUG)
endif()
if(VIS_BUILD_DAWN_RUNTIME)
  add_subdirectory(sdk/dawn)
  add_subdirectory(native/dawn)
endif()
