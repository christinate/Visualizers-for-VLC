name: Build
on:
  push:
    branches: [main]
    tags: ['v*']
  pull_request:
  workflow_dispatch:
permissions:
  contents: read
jobs:
  build:
    runs-on: windows-2022
    env:
      CMAKE_GENERATOR: Visual Studio 17 2022
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.12'
      - shell: pwsh
        run: |
          if (!(Test-Path "${env:ProgramFiles(x86)}\NSIS\makensis.exe")) { choco install nsis --no-progress -y }
      - run: python sdk/renderer-release.py
      - uses: actions/upload-artifact@v4
        with:
          name: Geiss64-candidate
          path: releases/*
