cmake_minimum_required(VERSION 3.24)
project(WinampVisualizerBridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(VISUALIZER_ID "geiss" CACHE STRING "Visualizer package ID")
if(VISUALIZER_ID STREQUAL "geiss")
  set(VISUALIZER_LABEL "Prism Flow bridge")
elseif(VISUALIZER_ID STREQUAL "terrain")
  set(VISUALIZER_LABEL "Rhythm Terrain bridge")
elseif(VISUALIZER_ID STREQUAL "beach")
  set(VISUALIZER_LABEL "Beach bridge")
elseif(VISUALIZER_ID STREQUAL "objects")
  set(VISUALIZER_LABEL "Objects bridge")
elseif(VISUALIZER_ID STREQUAL "undersea")
  set(VISUALIZER_LABEL "UnderSea bridge")
elseif(VISUALIZER_ID STREQUAL "coralcanyon")
  set(VISUALIZER_LABEL "CoralCanyon bridge")
else()
  set(VISUALIZER_LABEL "MilkDrop 2 bridge")
endif()
add_library(winamp_bridge MODULE bridge.cpp)
target_compile_definitions(winamp_bridge PRIVATE NOMINMAX VIS_ID=L"${VISUALIZER_ID}" VIS_LABEL="${VISUALIZER_LABEL}")
target_include_directories(winamp_bridge PRIVATE ../runtime/include)
target_link_libraries(winamp_bridge PRIVATE advapi32)
set_target_properties(winamp_bridge PROPERTIES PREFIX "" OUTPUT_NAME "vis_${VISUALIZER_ID}_bridge")
add_executable(bridge_probe bridge_probe.cpp)
target_compile_definitions(bridge_probe PRIVATE NOMINMAX)
target_include_directories(bridge_probe PRIVATE ../runtime/include)
