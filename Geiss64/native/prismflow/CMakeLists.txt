cmake_minimum_required(VERSION 3.24)
project(PrismFlowNativeCore LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Prism Flow requires a 64-bit target")
endif()
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
find_package(Threads REQUIRED)
add_library(prismflow_core STATIC wormhole.cpp motion.cpp selection.cpp background.cpp palette.cpp preset.cpp scheduler.cpp image_palette.cpp waveform.cpp frame_cpu.cpp)
target_link_libraries(prismflow_core PUBLIC Threads::Threads)
target_compile_features(prismflow_core PUBLIC cxx_std_20)
target_include_directories(prismflow_core PUBLIC .)
if(MSVC)
  target_compile_options(prismflow_core PRIVATE /utf-8 /fp:strict)
else()
  target_compile_options(prismflow_core PRIVATE -ffp-contract=off)
endif()
option(PRISMFLOW_BUILD_CORE_PROBE "Build the native/reference comparison probe" ON)
if(PRISMFLOW_BUILD_CORE_PROBE)
  add_executable(prismflow_core_probe core_probe.cpp)
  target_link_libraries(prismflow_core_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_core_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_motion_probe motion_probe.cpp)
  target_link_libraries(prismflow_motion_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_motion_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_background_probe background_probe.cpp)
  target_link_libraries(prismflow_background_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_background_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_palette_probe palette_probe.cpp)
  target_link_libraries(prismflow_palette_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_palette_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_preset_probe preset_probe.cpp)
  target_link_libraries(prismflow_preset_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_preset_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_scheduler_probe scheduler_probe.cpp)
  target_link_libraries(prismflow_scheduler_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_scheduler_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_image_palette_probe image_palette_probe.cpp)
  target_link_libraries(prismflow_image_palette_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_image_palette_probe PRIVATE ../../sdk/runtime/vendor)
  add_executable(prismflow_frame_probe frame_probe.cpp)
  target_link_libraries(prismflow_frame_probe PRIVATE prismflow_core)
  target_include_directories(prismflow_frame_probe PRIVATE ../../sdk/runtime/vendor)
  if(MSVC)
    target_compile_options(prismflow_core_probe PRIVATE /utf-8)
    target_compile_options(prismflow_motion_probe PRIVATE /utf-8)
    target_compile_options(prismflow_background_probe PRIVATE /utf-8)
    target_compile_options(prismflow_palette_probe PRIVATE /utf-8)
    target_compile_options(prismflow_preset_probe PRIVATE /utf-8)
    target_compile_options(prismflow_scheduler_probe PRIVATE /utf-8)
    target_compile_options(prismflow_image_palette_probe PRIVATE /utf-8)
  endif()
  enable_testing()
  find_program(NODE_EXECUTABLE node)
  if(NODE_EXECUTABLE)
    add_test(NAME prismflow_frame_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_frame.mjs"
      "$<TARGET_FILE:prismflow_frame_probe>" "${CMAKE_BINARY_DIR}/frame-reference-test")
    set_tests_properties(prismflow_frame_reference PROPERTIES TIMEOUT 180)
    add_test(NAME prismflow_image_palette_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_image_palette.mjs"
      "$<TARGET_FILE:prismflow_image_palette_probe>" "${CMAKE_BINARY_DIR}/image-palette-reference-test")
    set_tests_properties(prismflow_image_palette_reference PROPERTIES TIMEOUT 120)
    add_test(NAME prismflow_wormhole_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_native.mjs"
      "$<TARGET_FILE:prismflow_core_probe>" "${CMAKE_BINARY_DIR}/reference-test")
    set_tests_properties(prismflow_wormhole_reference PROPERTIES TIMEOUT 60)
    add_test(NAME prismflow_motion_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_motion.mjs"
      "$<TARGET_FILE:prismflow_motion_probe>" "${CMAKE_BINARY_DIR}/motion-reference-test")
    set_tests_properties(prismflow_motion_reference PROPERTIES TIMEOUT 180)
    add_test(NAME prismflow_background_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_background.mjs"
      "$<TARGET_FILE:prismflow_background_probe>" "${CMAKE_BINARY_DIR}/background-reference-test")
    set_tests_properties(prismflow_background_reference PROPERTIES TIMEOUT 180)
    add_test(NAME prismflow_palette_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_palette.mjs"
      "$<TARGET_FILE:prismflow_palette_probe>" "${CMAKE_BINARY_DIR}/palette-reference-test")
    set_tests_properties(prismflow_palette_reference PROPERTIES TIMEOUT 180)
    add_test(NAME prismflow_preset_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_preset.mjs"
      "$<TARGET_FILE:prismflow_preset_probe>" "${CMAKE_BINARY_DIR}/preset-reference-test")
    set_tests_properties(prismflow_preset_reference PROPERTIES TIMEOUT 240)
    add_test(NAME prismflow_scheduler_reference COMMAND "${NODE_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/prismflow_scheduler.mjs"
      "$<TARGET_FILE:prismflow_scheduler_probe>" "${CMAKE_BINARY_DIR}/scheduler-reference-test")
    set_tests_properties(prismflow_scheduler_reference PROPERTIES TIMEOUT 240)
  endif()
endif()
