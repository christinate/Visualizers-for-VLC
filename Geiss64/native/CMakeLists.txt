add_library(vis_geiss64 MODULE engine.cpp plugin.cpp)
set_target_properties(vis_geiss64 PROPERTIES PREFIX "" CXX_STANDARD 17)
target_include_directories(vis_geiss64 PRIVATE ../sdk/native ../sdk/runtime/include)
if(APPLE)
  target_link_libraries(vis_geiss64 PRIVATE "-framework OpenGL")
  set_target_properties(vis_geiss64 PROPERTIES SUFFIX ".dylib")
else()
  find_package(OpenGL REQUIRED)
  target_link_libraries(vis_geiss64 PRIVATE OpenGL::GL)
endif()
install(TARGETS vis_geiss64 LIBRARY DESTINATION plugins/geiss)
option(ENABLE_NATIVE_ENGINE_TESTS "Build the Geiss CPU lifecycle probe" OFF)
if(ENABLE_NATIVE_ENGINE_TESTS)
  add_executable(native_engine_probe engine.cpp ../tests/native_engine_probe.cpp)
  target_include_directories(native_engine_probe PRIVATE .)
endif()
