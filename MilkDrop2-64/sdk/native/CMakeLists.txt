cmake_minimum_required(VERSION 3.24)
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The native visualizer host requires a 64-bit target")
endif()
find_package(SDL2 REQUIRED)
find_package(Threads REQUIRED)
set(NATIVE_COMMON "${CMAKE_CURRENT_LIST_DIR}/../runtime/include")
set(NATIVE_JSON "${CMAKE_CURRENT_LIST_DIR}/../runtime/vendor")
add_executable(Visualizers host.cpp)
add_executable(VisualizersCtl control.cpp)
if(NOT APPLE)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
  target_link_libraries(VisualizersCtl PRIVATE OpenSSL::Crypto)
endif()
add_executable(VlcVisualizerLauncher launcher.cpp)
target_include_directories(Visualizers PRIVATE "${NATIVE_COMMON}" "${NATIVE_JSON}")
target_include_directories(VisualizersCtl PRIVATE "${NATIVE_JSON}")
target_link_libraries(Visualizers PRIVATE SDL2::SDL2 Threads::Threads ${CMAKE_DL_LIBS})
if(NOT APPLE)
  # Native release binaries must not retain the development SDL prefix.
  set_target_properties(Visualizers VisualizersCtl VlcVisualizerLauncher PROPERTIES
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/lib")
endif()
if(APPLE)
  enable_language(OBJCXX)
  target_sources(Visualizers PRIVATE settings_mac.mm)
  target_link_libraries(Visualizers PRIVATE "-framework AppKit" "-framework OpenGL")
else()
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  find_package(OpenGL REQUIRED)
  target_sources(Visualizers PRIVATE settings_qt.cpp)
  target_link_libraries(Visualizers PRIVATE Qt6::Widgets OpenGL::GL)
endif()
install(TARGETS Visualizers VisualizersCtl VlcVisualizerLauncher RUNTIME DESTINATION .)
