set(ENABLE_PLAYLIST OFF CACHE BOOL "" FORCE)
set(ENABLE_SYSTEM_PROJECTM_EVAL OFF CACHE BOOL "" FORCE)
set(ENABLE_SDL_UI OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(ENABLE_FAST_MATH OFF CACHE BOOL "" FORCE)
add_subdirectory(../vendor/projectm projectm)
add_library(vis_milkdrop64 MODULE plugin.cpp)
target_include_directories(vis_milkdrop64 PRIVATE ../sdk/native)
target_link_libraries(vis_milkdrop64 PRIVATE libprojectM::projectM)
set_target_properties(vis_milkdrop64 PROPERTIES PREFIX "" CXX_STANDARD 17 BUILD_WITH_INSTALL_RPATH TRUE)
if(APPLE)
  set_target_properties(vis_milkdrop64 PROPERTIES SUFFIX ".dylib" INSTALL_RPATH "@loader_path")
else()
  set_target_properties(vis_milkdrop64 PROPERTIES INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS vis_milkdrop64 LIBRARY DESTINATION plugins/milkdrop)
