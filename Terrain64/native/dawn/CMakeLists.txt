# Independent Terrain API 4 module, using this repository's pinned SDK.
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
add_library(terrain_dawn STATIC ../scene_dawn.cpp)
target_include_directories(terrain_dawn PUBLIC .. ../../sdk/runtime/include)
target_link_libraries(terrain_dawn PUBLIC visualizer_dawn visualizer_music)
add_library(terrain_native MODULE ../plugin_dawn.cpp)
target_include_directories(terrain_native PRIVATE ../../sdk/native)
target_link_libraries(terrain_native PRIVATE terrain_dawn)
set_target_properties(terrain_native PROPERTIES PREFIX "" OUTPUT_NAME "terrain_dawn" ARCHIVE_OUTPUT_NAME "terrain_dawn_import"
  BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../../native")
add_library(terrain_environment STATIC ../environment.cpp)
target_include_directories(terrain_environment PUBLIC .. ../../sdk/runtime/include)
target_link_libraries(terrain_environment PUBLIC visualizer_dawn visualizer_music)
foreach(identifier IN ITEMS beach objects undersea coralcanyon)
  string(TOUPPER "${identifier}" kind)
  add_library(${identifier}_native MODULE ../plugin_dawn.cpp)
  target_compile_definitions(${identifier}_native PRIVATE VIS_${kind}=1)
  target_include_directories(${identifier}_native PRIVATE ../../sdk/native)
  target_link_libraries(${identifier}_native PRIVATE terrain_environment)
  set_target_properties(${identifier}_native PROPERTIES PREFIX "" OUTPUT_NAME "${identifier}_dawn"
    ARCHIVE_OUTPUT_NAME "${identifier}_dawn_import" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "$ORIGIN/../../native")
endforeach()
