cmake_minimum_required(VERSION 3.24)
project(Terrain64 VERSION 0.5.0 LANGUAGES CXX)
option(VIS_BUILD_DAWN_RUNTIME "Build the native runtime and Terrain module from the pinned SDK" OFF)
if(UNIX)
  add_subdirectory(sdk/native)
  if(VIS_BUILD_DAWN_RUNTIME)
    add_subdirectory(sdk/dawn)
    add_subdirectory(native/dawn)
  endif()
  return()
endif()
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use a 64-bit MSVC build on Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory(sdk/runtime)
if(VIS_BUILD_DAWN_RUNTIME)
  add_subdirectory(sdk/dawn)
  add_subdirectory(native/dawn)
endif()
